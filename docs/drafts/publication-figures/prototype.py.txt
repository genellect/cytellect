cytellect-array-v1 hash equals the result's mask_sha256.

Run from the checkout (2026-11-19, main 7ff06b2b):

    .venv/bin/python docs/drafts/publication-figures/prototype.py.txt . <output directory outside the checkout>
"""

def saved_result():
    """The recorded comparison of nuclear NPM1 mean intensity over all nuclei (no positivity gate),
    and the figure the demo recorded for the same job."""
    journal = json.loads((SHOWCASE / "snapshot/journal.json").read_text(encoding="utf-8"))["entries"]
    for entry in journal:
        if (entry["method"] == "GET" and entry["scene"].startswith(SCENE)
                and re.fullmatch(r"/v1/jobs/[^/]+/common-statistics", entry["path"])):
            result = json.loads((SHOWCASE / "snapshot" / entry["file"]).read_text(encoding="utf-8"))
            if (result.get("metric") == "mean" and (result.get("channel") or {}).get("channel_id") == "npm1"
                    and result["region"]["region_set_id"] == "nuclei" and result.get("estimates")
                    and not (result.get("selection") or {}).get("channel_gate")
                    and result["spec"]["conditions"] == ORDER and result["spec"]["plot"]["kind"] == "superplot"):
                recorded = entry["path"].replace("/common-statistics", "/files/figure.svg")
                current = next(item["file"] for item in journal if item["path"] == recorded)
                return result, SHOWCASE / "snapshot" / current
    raise SystemExit("recorded comparison not found")


def array_hash(array):
    """cytellect-array-v1, as cytellect_analysis.regions._array_hash writes label images."""
    digest = hashlib.sha256()
    digest.update(f"cytellect-array-v1|<u4|{array.shape[0]},{array.shape[1]}|".encode("ascii"))
    digest.update(np.ascontiguousarray(array, dtype=np.dtype("<u4")).tobytes(order="C"))
    return digest.hexdigest()


class Seed:
    """Images and label arrays of the published initial state, found by their recorded hashes."""

    def __init__(self):
        self.zip = zipfile.ZipFile(SHOWCASE / "seed.zip")
        self.names = self.zip.namelist()
        self.tiffs = {hashlib.sha256(self.zip.read(name)).hexdigest(): name
                      for name in self.names if name.endswith(".tif")}

    def channel(self, field, index):
        name = self.tiffs[field["image_info"]["inputs"][f"ch{index}"]["sha256"]]
        return tifffile.imread(io.BytesIO(self.zip.read(name))).astype(float)

    def labels(self, field):
        for name in self.names:
            if field["field_id"] in name and name.endswith("labels.npy"):
                labels = np.load(io.BytesIO(self.zip.read(name)))
                if array_hash(labels) == field["mask_sha256"]:
                    return labels
        raise SystemExit("label image with the saved mask hash not found")


def swarm(values, width=.32):
    """Deterministic beeswarm: each point takes the smallest offset free of its neighbours."""
    values = np.asarray(values, float)
    step, near_y = .045, (np.ptp(values) or 1.0) * .028
    placed, offsets = [], np.zeros(len(values))
    for index in np.argsort(values, kind="stable"):
        y = values[index]
        taken = [px for px, py in placed if abs(py - y) < near_y]
        for k in range(200):
            x = (k + 1) // 2 * step * (1 if k % 2 else -1)
            if all(abs(x - px) >= step * .95 for px in taken):
                break
        offsets[index] = float(np.clip(x, -width, width))
        placed.append((offsets[index], y))
    return offsets


result, recorded_figure = saved_result()
rows = [row for row in result["plot_data"] if row["selection_status"] == "selected"]
units = result["unit_summary"]
fields = {field["field_id"]: field for field in result["source_fields"]}
runs = sorted({unit["pair"] for unit in units})
run_color = dict(zip(runs, OKABE_ITO))
group_mean = {c: float(np.mean([u["value"] for u in units if u["condition"] == c])) for c in ORDER}
seed = Seed()
channel_index = {c["channel_id"]: i for i, c in enumerate(next(iter(fields.values()))["image_info"]["channels"])}


def representative(condition):
    """The field whose median is closest to the condition's mean of runs (ties: field order)."""
    candidates = []
    for position, (fid, field) in enumerate(fields.items()):
        values = [row["value"] for row in rows if row["field_id"] == fid]
        if field["metadata"]["condition"] == condition and len(values) >= 3:
            candidates.append((abs(float(np.median(values)) - group_mean[condition]), position, fid))
    return min(candidates)[2]


def crop(fid, labels, size=200):
    """A square centred on the measured nucleus closest to the condition's mean of runs."""
    condition = fields[fid]["metadata"]["condition"]
    chosen = min((abs(row["value"] - group_mean[condition]), row["region_id"])
                 for row in rows if row["field_id"] == fid)[1]
    ys, xs = np.nonzero(labels == chosen)
    y0 = int(np.clip(int(ys.mean()) - size // 2, 0, labels.shape[0] - size))
    x0 = int(np.clip(int(xs.mean()) - size // 2, 0, labels.shape[1] - size))
    return slice(y0, y0 + size), slice(x0, x0 + size)


shown = {}
for condition in ORDER:
    fid = representative(condition)
    labels = seed.labels(fields[fid])
    box = crop(fid, labels)
    shown[condition] = {"field": fid, "labels": labels[box],
                        "npm1": seed.channel(fields[fid], channel_index["npm1"])[box],
                        "hoechst": seed.channel(fields[fid], channel_index["hoechst"])[box]}
# One linear display range per channel, shared by both conditions; recorded, never per-image.
display = {name: tuple(float(v) for v in np.percentile(
    np.concatenate([shown[c][name].ravel() for c in ORDER]), [.5, 99.5])) for name in ("npm1", "hoechst")}


def scaled(condition, name):
    low, high = display[name]
    return np.clip((shown[condition][name] - low) / (high - low), 0, 1)

    "ytick.labelsize": 6.5, "legend.fontsize": 6, "axes.linewidth": .5, "xtick.major.width": .5,
    "ytick.major.width": .5, "xtick.major.size": 2, "ytick.major.size": 2, "lines.linewidth": .6,
    "svg.fonttype": "none", "pdf.fonttype": 42, "axes.spines.top": False, "axes.spines.right": False,
})

        axes.imshow(picture, cmap="gray" if column == 0 else None, vmin=0, vmax=1, interpolation="nearest")

        for spine in axes.spines.values():
            spine.set_visible(False)
        if column == 0:
            for region in measured:
                filled = ndimage.binary_fill_holes(shown[condition]["labels"] == region).astype(float)
                for contour in measure.find_contours(filled, .5):
                    axes.plot(contour[:, 1], contour[:, 0], color="#ffd400", lw=.45)
            axes.set_ylabel(condition.replace(" 10 µM", "\n10 µM"), labelpad=3)
        if row == 0 and column == 0:
            axes.set_title("NPM1", fontsize=6.5, pad=2.5)
        if row == 0 and column == 1:
            axes.text(.47, 1.035, "Hoechst", color="#1a9e3a", transform=axes.transAxes, ha="right", va="bottom")
            axes.text(.53, 1.035, "NPM1", color="#c0119d", transform=axes.transAxes, ha="left", va="bottom")
        if row == 1 and column == 1:
            side, bar = picture.shape[0], 10 / pixel_um
            axes.plot([side - 12 - bar, side - 12], [side - 14, side - 14], color="white", lw=1.6, solid_capstyle="butt")

figure.text(.012, .94, "a", fontsize=8, fontweight="bold", va="top")

# Descriptive comparison (test "none"): the saved per-run values only, no interval and no p value.

elif higher * 2 > len(saved_runs):

else:
    note = f"Lower in {len(saved_runs) - higher} of {len(saved_runs)} runs"

axes.text(.5, 4840, note, ha="center", va="bottom", fontsize=6)
axes.legend(handles=[Line2D([], [], marker="o", ls="", color=run_color[run], markeredgecolor="black",
                            markeredgewidth=.4, markersize=3.6, label=f"Run {index + 1}")
